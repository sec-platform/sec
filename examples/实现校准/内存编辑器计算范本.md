## 1. 完整范围、规模和真实缺项

“完整”指列出的作者模块、互相调用的公开定义和本节功能路径没有省略主体。它不把宿主声明当成实现，也不使用`...`、空TODO或未定义的`createIDE`来隐藏关键功能。应用使用者只改两个入口；功能维护者进入`features`和领域定义；底层算法维护者进入`search/literal`等实际主体。这些文件可在同一项目起步或按已有包机制复用，没有强制28个独立包/进程。

当前输入域是有限、去重、已经取得的普通文本文件。`Text`按Unicode标量解释，不隐式规范化；本宿主画像选择UTF-8无BOM并原样保留CR/LF。非该编码、损坏文本、笔记本、任意虚拟/远端写入不能静默解码或剔除后称全部支持。原更大工程仍可通过自己已有的合格适配处理它们。

搜索是非空字面量、大小写敏感、不重叠匹配。范围中的每个文件都保留，包括零命中文件。搜索函数完整返回这个固定范围；取消、资源超限或提供者失败不能把部分值伪装完整结果。性能算法是可审查的朴素参考设计，非生产最优实现。示例中500个文档、100次历史或应用覆盖的200只是明确可修改参数，不是科学最优阈值；真实内存/时间还由宿主预算限制，不能靠文档数独自防止大文本耗尽。

`Int`仍是原任意精度整数语义；目标TS不能把修订号或文本位置偷偷降低为会舍入的Number。文本原语的索引按标量，Monaco等采用其他坐标的宿主必须准确转换，并拒绝落在不合法边界的输入，不能直接拼接两种offset。

```sec
  match left { case [] => right; case [head, ..tail] => [head, ..append(tail, right)]; };
export fn reverseInto<T>(items: List<T>, acc: List<T>) -> List<T> !{} =
```

```sec
import "./model.sec";
import "../shared/seq.sec";
export type Error = ReadOnly() | Stale(Int, Int) | NothingToUndo() | NothingToRedo() | InvalidHistoryLimit();
export fn replace(doc: model.Document, expected: Int, value: Text, historyLimit: Int)
  -> Result<model.Document, Error> !{} =
  if !doc.writable then Result.Err(Error.ReadOnly())
  else if doc.revision != expected then Result.Err(Error.Stale(expected, doc.revision))
  else if historyLimit < 0 then Result.Err(Error.InvalidHistoryLimit())
  else if doc.text == value then Result.Ok(doc)
  else Result.Ok(doc with {
    text: value, revision: doc.revision + 1,
    undo: seq.take([doc.text, ..doc.undo], historyLimit), redo: []
  });
export fn undo(doc: model.Document, historyLimit: Int) -> Result<model.Document, Error> !{} =
  if !doc.writable then Result.Err(Error.ReadOnly()) else if historyLimit < 0
  then Result.Err(Error.InvalidHistoryLimit()) else match doc.undo {
    case [] => Result.Err(Error.NothingToUndo());
    case [previous, ..rest] => Result.Ok(doc with {
      text: previous, revision: doc.revision + 1, undo: rest,
      redo: seq.take([doc.text, ..doc.redo], historyLimit)
    });
  };
export fn redo(doc: model.Document, historyLimit: Int) -> Result<model.Document, Error> !{} =
  if !doc.writable then Result.Err(Error.ReadOnly()) else if historyLimit < 0
  then Result.Err(Error.InvalidHistoryLimit()) else match doc.redo {
    case [] => Result.Err(Error.NothingToRedo());
    case [next, ..rest] => Result.Ok(doc with {
      text: next, revision: doc.revision + 1, redo: rest,
      undo: seq.take([doc.text, ..doc.undo], historyLimit)
    });
  };
```

```sec
import "./engine.sec";
export type Phase = Idle() | Running(Int, Int) | Ready(Int, engine.ResultSet) | Failed(Text);
export type State = { epoch: Int, needle: Text, replacement: Text, phase: Phase };
export fn initial() -> State !{} = {epoch: 0, needle: "", replacement: "", phase: Phase.Idle()};
export fn changed(state: State, needle: Text) -> State !{} =
  state with {epoch: state.epoch + 1, needle: needle, phase: Phase.Idle()};
export fn invalidate(state: State) -> State !{} =
  state with {epoch: state.epoch + 1, phase: Phase.Idle()};
export fn accept(state: State, session: Text, corpus: Int, epoch: Int,
  resultSet: engine.ResultSet) -> State !{} =
  match state.phase {
    case Phase.Running(expected, expectedCorpus) =>
      if epoch == expected && expected == state.epoch && corpus == expectedCorpus
         && resultSet.scope.session == session && resultSet.scope.corpus == corpus
         && resultSet.query.needle == state.needle
      then state with {phase: Phase.Ready(epoch, resultSet)} else state;
    case _ => state;
  };
```

### `replace/plan.sec`

```sec
import "../workspace/resource.sec";
import "../search/literal.sec";
import "../search/engine.sec";
import "../platform/text.sec";
import "../shared/seq.sec";
export type FileEdit = { id: resource.Id, expected: Int, before: Text, after: Text };
export type Plan = { session: Text, corpus: Int, files: List<FileEdit> };
export type Error = ReadOnly(resource.Id) | UnknownSelection(resource.Id) | DuplicateSelection(resource.Id)
  | InvalidMatch(resource.Id) | EmptySelection();
fn compose(value: Text, spans: List<literal.Span>, replacement: Text, at: Int) -> Text !{} =
  match spans {
    case [] => text.slice(value, at, text.length(value));
    case [head, ..tail] => text.concat(text.slice(value, at, head.start),
      text.concat(replacement, compose(value, tail, replacement, head.end)));
  };
fn validSpans(value: Text, needle: Text, spans: List<literal.Span>, after: Int) -> Bool !{} =
  match spans {
    case [] => true;
    case [head, ..tail] => after <= head.start && head.start < head.end
      && head.end <= text.length(value)
      && text.slice(value, head.start, head.end) == needle
      && validSpans(value, needle, tail, head.end);
  };
fn find(files: List<engine.FileResult>, id: resource.Id) -> Option<engine.FileResult> !{} =
  match files { case [] => Option.None(); case [head, ..tail] =>
    if resource.same(head.source.id, id) then Option.Some(head) else find(tail, id);
  };
fn selected(files: List<engine.FileResult>, ids: List<resource.Id>, needle: Text,
  replacement: Text) -> Result<List<FileEdit>, Error> !{} =
  match ids {
    case [] => Result.Ok([]);
    case [id, ..tail] =>
      if seq.any(tail, (other: resource.Id) => resource.same(id, other))
      then Result.Err(Error.DuplicateSelection(id)) else match find(files, id) {
        case Option.None() => Result.Err(Error.UnknownSelection(id));
        case Option.Some(found) => if !found.source.writable then Result.Err(Error.ReadOnly(id))
          else if !validSpans(found.source.text, needle, found.spans, 0)
          then Result.Err(Error.InvalidMatch(id)) else match selected(files, tail, needle, replacement) {
            case Result.Err(error) => Result.Err(error);
            case Result.Ok(rest) =>
              let next = compose(found.source.text, found.spans, replacement, 0);
              if next == found.source.text then Result.Ok(rest)
              else Result.Ok([{id: id, expected: found.source.revision,
                before: found.source.text, after: next}, ..rest]);
          };
      };
  };
export fn make(resultSet: engine.ResultSet, ids: List<resource.Id>, replacement: Text)
  -> Result<Plan, Error> !{} =
  match ids {
    case [] => Result.Err(Error.EmptySelection());
    case [head, ..tail] => match selected(resultSet.files, ids, resultSet.query.needle, replacement) {
      case Result.Err(error) => Result.Err(error);
      case Result.Ok(edits) => Result.Ok({session: resultSet.scope.session,
        corpus: resultSet.scope.corpus, files: edits});
    };
  };
```

```sec
import "./plan.sec";
import "../documents/model.sec" as documents;
import "../documents/edit.sec";
import "../workspace/resource.sec";
export type Error = WrongScope() | Missing(resource.Id) | Changed(resource.Id) | Rejected(resource.Id, edit.Error);
fn applyFiles(docs: List<documents.Document>, edits: List<plan.FileEdit>, historyLimit: Int)
  -> Result<List<documents.Document>, Error> !{} =
  match edits {
    case [] => Result.Ok(docs);
    case [head, ..tail] => match documents.find(docs, head.id) {
      case Option.None() => Result.Err(Error.Missing(head.id));

        then Result.Err(Error.Changed(head.id)) else match edit.replace(doc, head.expected, head.after, historyLimit) {
          case Result.Err(error) => Result.Err(Error.Rejected(head.id, error));

  historyLimit: Int) -> Result<List<documents.Document>, Error> !{} =

  else applyFiles(docs, preview.files, historyLimit);
```

```sec
export type Request = { id: Int, resource: resource.Id, revision: Int,
```

```sec
        "Document disappeared while save was pending")), notice: "Needs reconciliation"};
      case Option.Some(doc) => if writtenText != request.text
        then {docs: docs, phases: saving.set(phases, saving.Phase.Uncertain(request,
          "Save participant changed the promised bytes")), notice: "Saved bytes differ; reconcile"}
        else {docs: documents.put(docs, doc with {savedText: writtenText, diskStamp: stamp}),
          phases: saving.remove(phases, request.resource), notice: "Saved snapshot; newer edits remain"};
    };
    case saving.Receipt.Refused(reason) => {docs: docs,
      phases: saving.set(phases, saving.Phase.Failed(request, reason)), notice: reason};
    case saving.Receipt.Unknown(reason) => {docs: docs,
      phases: saving.set(phases, saving.Phase.Uncertain(request, reason)), notice: reason};
  };
export fn accept(docs: List<documents.Document>, phases: List<saving.Phase>, id: resource.Id,
  requestId: Int, receipt: saving.Receipt) -> Change !{} = match saving.find(phases, id) {
    case Option.None() => {docs: docs, phases: phases, notice: "No matching save"};
    case Option.Some(phase) => let request = saving.ofPhase(phase);
      if request.id != requestId then {docs: docs, phases: phases, notice: "Old save receipt ignored"}
      else match phase {
        case saving.Phase.Failed(_, _) => {docs: docs, phases: phases, notice: "Terminal save already settled"};
        case _ => complete(docs, phases, request, receipt);
      };
  };
```

```sec
    case Option.None() => if seq.size(model.documents) >= model.settings.maxLoadedDocuments

    case Result.Err(error) => match error {
      case edit.Error.ReadOnly() => app.note(model, "Read-only document");
      case edit.Error.Stale(_, _) => app.note(model, "Editor revision is stale");
      case edit.Error.NothingToUndo() => app.note(model, "Nothing to undo");
      case edit.Error.NothingToRedo() => app.note(model, "Nothing to redo");
      case edit.Error.InvalidHistoryLimit() => app.note(model, "Invalid history limit");

    case Result.Ok(doc) => match documents.find(model.documents, doc.id) {
      case Option.None() => app.note(model, "Document disappeared");
      case Option.Some(previous) => if previous.revision == doc.revision
        then app.stay(model) else app.changed(model, documents.put(model.documents, doc));
    };
  };

    case Option.None() => app.note(model, "Document not loaded");
    case Option.Some(doc) => accept(model, edit.replace(doc, expected, value, model.settings.historyLimit));
  };

  match documents.find(model.documents, id) {
    case Option.None() => app.note(model, "Document not loaded");
    case Option.Some(doc) => accept(model, edit.undo(doc, model.settings.historyLimit));
  };
export fn redo(model: app.Model, id: resource.Id) -> app.Transition !{} =
  match documents.find(model.documents, id) {
    case Option.None() => app.note(model, "Document not loaded");
    case Option.Some(doc) => accept(model, edit.redo(doc, model.settings.historyLimit));
  };
export fn activate(model: app.Model, id: resource.Id) -> app.Transition !{} =
  match documents.find(model.documents, id) {
    case Option.None() => app.note(model, "Document not loaded");
    case Option.Some(_) => app.stay(model with {active: Option.Some(id)});
  };
```

```sec
import "../product/model.sec" as app;
import "../product/messages.sec";
import "../search/query.sec";
import "../search/state.sec" as searching;
import "../search/snapshot.sec";
import "../search/engine.sec";

import "../replace/apply.sec" as replacement;
import "../workspace/resource.sec";
export fn needle(model: app.Model, value: Text) -> app.Transition !{} =
  app.stay(model with {search: searching.changed(model.search, value), preview: Option.None()});
export fn replacementText(model: app.Model, value: Text) -> app.Transition !{} =
  app.stay(model with {search: model.search with {replacement: value}, preview: Option.None()});
export fn select(model: app.Model, ids: List<resource.Id>) -> app.Transition !{} =
  app.stay(model with {selected: ids, preview: Option.None()});
export fn begin(model: app.Model) -> app.Transition !{} = match query.make(model.search.needle) {
  case Result.Err(_) => app.note(model, "Search text is empty");
  case Result.Ok(q) =>
    let epoch = model.search.epoch + 1;
    let scope = snapshot.capture(model.session, model.corpus, model.documents);
    {model: model with {search: model.search with {
       epoch: epoch, phase: searching.Phase.Running(epoch, model.corpus)}, preview: Option.None()},
     commands: [messages.Command.Search(epoch, scope, q)]};
};
export fn finish(model: app.Model, epoch: Int, found: engine.ResultSet) -> app.Transition !{} =
  app.stay(model with {search: searching.accept(model.search, model.session, model.corpus, epoch, found)});
export fn fail(model: app.Model, epoch: Int, reason: Text) -> app.Transition !{} =
  match model.search.phase {
    case searching.Phase.Running(expected, _) => if epoch == expected
      then app.stay(model with {search: model.search with {phase: searching.Phase.Failed(reason)}})
      else app.stay(model);
    case _ => app.stay(model);
  };
export fn preview(model: app.Model) -> app.Transition !{} = match model.search.phase {
  case searching.Phase.Ready(_, found) => match plan.make(found, model.selected, model.search.replacement) {
    case Result.Err(_) => app.note(model, "Selection is empty, stale, duplicated, invalid or read-only");
    case Result.Ok(value) => app.stay(model with {preview: Option.Some(value), notice: "Preview ready"});
  };
  case _ => app.note(model, "A current completed search is required");
};
export fn apply(model: app.Model) -> app.Transition !{} = match model.preview {
  case Option.None() => app.note(model, "No current replacement preview");
  case Option.Some(value) => match replacement.apply(model.documents, model.session, model.corpus,
    value, model.settings.historyLimit) {
      case Result.Err(_) => app.note(model, "Preview cannot be applied to current documents");
      case Result.Ok(docs) => if docs == model.documents
        then app.note(model, "No text changes")
        else let next = app.changed(model, docs);
          next with {model: next.model with {notice: "Applied to editor models; save is separate"}};
    };
};
```

## 3. 哪些已经有主体，哪些必须交给真实宿主

### 三个纯叶原语

`text.length`给出Unicode标量数；`slice(value,start,end)`在已声明合法区间返回半开切片，越界不截断成看似成功；`concat`保留准确标量序列。三个原语都是纯值操作，其目标实现必须保持内容、数值与资源错误。不能因签名为`!{}`就假定不耗时、不分配、不会超预算。当前模块没有声明正则、文件IO、网络、剪贴板、GUI或任意“魔法工作台原语”。

### Program的真实宿主接合

| 交接 | 准确输入与结果 | 行为、失败和拥有者 |
|---|---|---|
| 构造Program | `apps.*.product()`返回`Result<Program,settings.Error>` | 普通运行调用，校验配置后返回真实函数值；不是静态初始化执行宿主，也不是新框架关键字 |
| 打开实例 | 唯一的当前`session:Text`，调用`Program.initial(session)` | 宿主分配并绑定一个活动实例；每次重开取得不同session；会话名不授予权限 |
| 事件采用 | `Program.update(session,event,current)`返回`Transition` | 宿主串行处理本实例消息并一次安装纯返回model；错session/closed返回原model。先完成安装和命令登记，再启动可能同步完成的外部工作，不在计算纯值中发IO |
| 执行搜索 | `Command.Search(epoch,Scope,Query)`，调用`Program.search` | 可同线程或合格任务执行；回到原session的Finished/Failed。Scope不可改，完整结束才交ResultSet；结果晚到只占自身资源，不覆新查询。允许取消旧搜索但必须真正结算其资源 |
| 写磁盘 | `Command.Save(Request)`及原session | 条件使用`expectedDisk`，实际权限、编码、资源身份和最后写入条件由真实写边界检查。同一session/request.id固定同一载荷，去重/查回；只有先读后写且竞争者可插入的提供者不满足强版本条件，须报告能力缺项或另选明确较弱画像；不能把纯Command当成已写 |
| 写入结果 | `Written(stamp,text)`、`Refused(reason)`或`Unknown(reason)` | Written仅声明本次准确文本及实际新版本已按该画像写入，不承诺跨文件事务或断电耐久；Refused须已证明无效果且旧尝试不会晚到作用；否则Unknown。沿既有恢复规则，不按超时猜Refused |
| 屏幕 | `Program.view(model)`返回`Screen` | 被动投影，未执行布局引擎/DOM/绘制；宿主映射固定控件并提交准确Event，不通过notice字符串执行脚本 |
| 命令 | `commands`和`invoke(session,Command,model)` | 本例闭合静态命令集；打开式插件注册、跨进程扩展激活不是枚举加一项便全部完成 |
| 停止 | `Command.Stop()` | 源模型已禁止新编辑；宿主撤销并排空搜索/渲染任务，释放引用后才称实例退出。它不能杀其他会话或忘记未决保存 |

接纳命令失败也必须成为当前任务的可观察失败；不能安装一个永远Pending却没有运行者或恢复记录的保存。Unknown后宿主保留原请求并按真实操作查回；准确结算后仍通过同一session/requestId的SaveFinished返回，不重放Save。暂不可判定就保持Uncertain并显示相应等待/协调状态，不伪造一个自动恢复时限。外部不可信数据先按已绑定类型及资源/快照合同解码；纯函数签名不替宿主验证来源、命中覆盖或写入许可。异步保存/搜索回调中的session来自启动时捕获值，不是收到回调后再读取`latest.session`。已封存回执与当前显示资格不同；本源的会话检查不能替代真实驱动对写权限和旧尝试的保护。

本例根状态是可替换的不可变值，整批replace先计算新文档列表；任一分支失败会丢弃尚未采用的新值。因此可对**本实例模型**一次安装，不意味所有磁盘文件原子写入。实际多线程共享模型、跨节点协同或动态对象身份需选对应同步设计，不自动从这个纯函数取得。

“dirty=false”只表示当前文本等于本程序最后确认保存的文本；外部进程随后改磁盘时，不得把这个标记说成实时磁盘一致。新增外部变更监听应接成具有版本和冲突处理的事件，而非覆盖脏模型。源中的Failed保存仅在真实终态拒绝合同下可发新尝试；Unknown保留原请求，不能换ID盲重发。

崩溃恢复不在这28个内存模块里实现。需要退出后恢复未决保存时，真实宿主应把session、意图、载荷、去重范围和回执写入既有持久协议；没有该能力就不得宣称支持崩溃后恰好一次。恢复还要独立取得当前写权，不能执行旧队列当重建UI。本例不以扩展内存Model偷偷承诺磁盘日志已经存在。

### 屏幕不是一句“展示即可”

`Screen.documents`逐行显示label、dirty、writable、selected，选择改变只提交该行实际Id；打开文档提交Activate。`Screen.editor`映射现有文本编辑控件，保留输入法、选区和焦点，提交Edited携带该显示版本的expectedRevision；只读时禁编辑但仍在更新函数核验。`Search.needle/replacement`分别提交文本事件；搜索按钮发SearchRequested；预览显示每文件before/after的真实差异；应用和保存是两个独立动作，不以“应用”按钮暗中写盘。notice可显示错误但不能替类型化状态或权限。

键盘命令/辅助技术控件与点击进入同一事件，不单独写model。状态更新不强制销毁重建编辑器导致输入法或焦点丢失；宿主保留以Resource.Id绑定的视图实例。文档关闭/取消的提示从dirty及未决保存判断，DiscardAndClose只丢尚未保存编辑，不撤销已写入磁盘。当前`canPreview/canApply`仅给常见可用性，真实动作仍可能因未知选择、只读或失效拒绝，按钮不是授权证明。

这里已经固定视图信息、控件语义及事件映射，但没有提供完整产品CSS、品牌、字体、屏幕阅读器实测、Web/桌面两套渲染器。它们是可继续实现的宿主工作，不把一份Screen记录叫做最终视觉设计，也不把它变成SEC必须图形化开发的决定。

## 4. 从源码复算行为，不以目录存在代替正确性

以下是按函数定义的书面推导，不是产品运行成绩。

**一次替换与第二次修改。** 文档A为`cat cat`，修订0，已保存值相同。搜索`cat`得到`[0,3)`和`[4,7)`；替换`dog`的Plan保存前像`cat cat`、后像`dog dog`及修订0。应用得到修订1、undo首项`cat cat`，savedText不变。再次采用原计划会因corpus/修订条件拒绝，不能重复套用。

**保存期间继续编辑。** 对修订1发保存请求1，冻结`dog dog`；用户随后改成`dog!`，修订2。旧Written确认保存的是`dog dog`，结算只更新savedText和diskStamp，文本保持`dog!`，所以仍dirty。若先undo回`cat cat`，修订仍增加，不复用0冒充旧快照未变化。

**旧异步结果与重开。** 查询A后换查询B会增加epoch；A完成不会安装。相同文本换回A也取得新epoch。整个工作台重开时换session；即便request.id或epoch从1重新开始，旧回调带旧session，被根update拒绝，不能靠数值相同穿越实例。

**非重叠和标量坐标。** `aaaa`查`aa`得到`[0,2),[2,4)`，不是同时替换`[1,3)`；`a😀a`查`😀`得到标量区间`[1,2)`。UTF-16宿主必须转换为自己的真实区间，不把2当作其结束代码单元。空needle明确报错，避免零长无限循环。

**不完整范围。** 未加载的文件没有在Scope中；这次只称完整搜索已加载集合。完整磁盘搜索、忽略规则、未保存模型覆盖和远程Provider由原Code-OSS范本及适配承担，本例没有把它们假装发生。

**选择与物理保存。** 重复Id、不存在Id和选中只读文档有各自错误。选择中零命中/后像不变的文档不造成虚假编辑。内存整批成立仍可能一个文件写入成功另一个失败；当前设计分项保存，绝不将它报告为跨文件事务。

## 5. 第二次开发：改产品差异与更换算法主体

### 一份完整的新应用入口

用户要求同一产品仅保留40次历史而不是200次，不改词法、不改Product类型、不重新抄所有模块。`apps/desktop.sec`的完整后继为：

```sec
import "../product/definition.sec";
import "../product/settings.sec";
export fn product() -> Result<definition.Program, settings.Error> !{} =
  definition.define(settings.standard with {title: "Workspace Editor", historyLimit: 40});
```

这会改变后续新实例的配置；不会偷偷热改旧实例的历史。需要主动缩短活动实例已保存的历史时，还需显式配置变更事件及资源/撤销影响，不因重新生成源码自动执行。

### 一份完整的新匹配器主体

本版朴素扫描对连续命中在返回链上构造列表。可以在保持相同非重叠结果的前提下改为累积反转，完整替换`search/literal.sec`如下：

```sec
import "../platform/text.sec";
import "../shared/seq.sec";
import "./query.sec";
export type Span = { start: Int, end: Int };
fn scan(value: Text, needle: Text, at: Int, width: Int, total: Int,
  reversed: List<Span>) -> List<Span> !{} =
  if at + width > total then seq.reverse(reversed)
  else if text.slice(value, at, at + width) == needle
  then scan(value, needle, at + width, width, total, [{start: at, end: at + width}, ..reversed])
  else scan(value, needle, at + 1, width, total, reversed);
export fn find(value: Text, q: query.Query) -> List<Span> !{}
  require q.needle != ""
= scan(value, q.needle, 0, text.length(q.needle), text.length(value), []);
```

公共签名、Span和非重叠含义相同；改变私有算法只重算相交编译和行为检查，其他消费者不手工同步一份影子算法。尾递归形式**不证明所选TS生成方法已进行尾调用降低**；没有该能力仍可能用栈，必须准确报告。这个变化也不自动降低每次slice的成本，不冒称性能提高倍数。

若需求改成重叠匹配、正则、忽略大小写或近似Unicode匹配，不能只换这个函数然后宣称替换同义：Query与flags、捕获和替换构造、位置映射、预览有效性、选择/冲突及检验都受影响。重叠命中没有唯一的批量替换含义，应选择精确非重叠子集、规定真实冲突算法，或者仅支持查看而拒绝批量应用；不能凭匹配数增加说功能更完备。

## 6. 目标源长什么样，怎样继续改

下段是记录更新的**完整有限TS目标形状设计**，没有声称由SEC编译器实际生成。对象仅是已验证的中立值表示，不是任意带getter/proxy的原生对象；有限ExampleSettings使用者不能据此推论全部Document表示。

```ts
type ExampleSettings = Readonly<{
  title: string;
  historyLimit: bigint;
  maxLoadedDocuments: bigint;
}>;
export function withHistoryLimit(
  original: ExampleSettings,
  limit: bigint,
): ExampleSettings {
  const base = original;
  const nextLimit = limit;
  return {
    title: base.title,
    historyLimit: nextLimit,
    maxLoadedDocuments: base.maxLoadedDocuments,
  };
}
```

对应作者表达式是`original with {historyLimit: limit}`。实际一般降低还要保留该名义记录表示的品牌、字段策略和精确来源；这段局部形状不替代整个目标画像。`Int`用BigInt或另一合格表示，不凭Number暂未溢出偷换语义。Text的长度/切片不能直接用JS的代码单元索引。

受管目标修改若只改显式更新的limit，可以回到该作者字段；目标合成的`title: base.title`是保留语义，不是作者又声明了一份title。目标编辑改成新行为、越过不透明表示、调整可观察求值顺序或者来源多解时保留目标草稿/冲突，再选择真实源变更，不自动把任意TS解释成合法with。
